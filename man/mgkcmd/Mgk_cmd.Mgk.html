type comp_op = string

composite operator

val over : comp_op

default operator, overlays the source image over the destination image

val multiply : comp_op

multiplies the source image with the destination image

val screen : comp_op

multiplies the inverse of the images and then inverses the result

val overlay : comp_op

combines Multiply and Screen

val darken : comp_op

takes the darker pixel value

val lighten : comp_op

takes the lighter pixel value

val add : comp_op

adds pixel values

val subtract : comp_op
